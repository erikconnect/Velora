Elevation Logic

// OKLCH Elevation Function
@function set-elevation($level) {
  $base-l: 13%;
  $base-c: 0.015;
  $base-h: 160;

  $delta: (
    0: 0%,
    1: 2.5%,
    2: 5.0%,
    3: 8.5%,
    4: 14.0%
  );

  @return oklch(
    $base-l + map-get($delta, $level), 
    $base-c + ($level * 0.003), 
    $base-h
  );
}
