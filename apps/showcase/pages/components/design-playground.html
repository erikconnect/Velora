Theme switching formula
/* Programmatic theme switching */
document.documentElement.setAttribute('data-theme', 'dark');          // → Noir surfaces
document.documentElement.setAttribute('data-theme', 'light');         // → Earth surfaces
document.documentElement.setAttribute('data-editorial-theme', 'noir');  // → Noir editorial
document.documentElement.setAttribute('data-editorial-theme', 'earth'); // → Earth editorial

/* Or via CSS */
html[data-theme="dark"]  { /* dark token overrides */ }
html[data-theme="light"] { /* light token overrides */ }
